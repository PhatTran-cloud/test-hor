from hotel_scene_base import *

class Section3Scene(HotelScene):
    def construct(self):
        self.setup_layout("ONE INFINITE BUS: THE EVEN-ODD ZIPPER", [], number=3)
        self.chapter("Old guests keep the even rooms. New guests take the odd rooms.")
        xs=[-5.8+i*1.28 for i in range(10)]
        doors=VGroup(*[room_front(i+1,[x,.35,0],width=1.0,height=1.15,color=DIM) for i,x in enumerate(xs)])
        old=VGroup(*[guest(1.0,BLUE).move_to([-4.6+i*1.45,1.55,0]) for i in range(5)]).set_z_index(5)
        old_tag=label("EXISTING GUESTS",[-4.3,2.0,0],BLUE,18)
        bus=bus_shape([6.0,-1.7,0])
        self.play(FadeIn(doors),FadeIn(old),FadeIn(old_tag),run_time=1.5)
        self.play(bus.animate.shift(LEFT*1.7),run_time=1.5)
        bus_tag=label("1, 2, 3, ... NEW GUESTS",[4.1,-2.55,0],PINK,18)
        self.play(FadeIn(bus_tag),run_time=.6)
        formula=MathTex(r"n\longmapsto 2n",color=BLUE,font_size=37).move_to([-2.5,-1.55,0])
        self.wait(1)
        self.play(Write(formula),run_time=1)
        self.play(*[MoveAlongPath(p,travel_path(p.get_center(),[xs[2*i+1],.1,0],-.55)) for i,p in enumerate(old)],run_time=3)
        self.play(*[doors[i][2].animate.set_color(BLUE) for i in range(1,10,2)],run_time=.7)
        self.wait(1.2)
        self.play(LaggedStart(*[doors[i][1].animate.set_fill("#1B4934",opacity=1) for i in range(0,10,2)],lag_ratio=.13),run_time=1.3)
        newcomers=VGroup(*[guest(1.0,PINK).move_to([3.65,-1.65,0]) for _ in range(5)]).set_z_index(5)
        self.play(*[MoveAlongPath(p,travel_path(p.get_center(),[xs[2*i],.1,0],.55)) for i,p in enumerate(newcomers)],run_time=3.5)
        self.play(*[doors[i][2].animate.set_color(PINK) for i in range(0,10,2)],run_time=.7)
        odd_rule=MathTex(r"k\longmapsto 2k-1",color=PINK,font_size=32).move_to([-2.5,-2.5,0])
        self.play(Write(odd_rule),run_time=1)
        self.play(Indicate(doors,color=GOLD),run_time=1)
        self.finish_slot()
