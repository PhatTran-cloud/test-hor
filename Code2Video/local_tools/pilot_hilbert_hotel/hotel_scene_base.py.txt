from pilot_scene_base import *

GOLD = "#F2C14E"
GREEN = "#75D69A"
BLUE = "#6EA8FE"
PINK = "#F08AB4"
PURPLE = "#B39DDB"
PANEL = "#0D1B2A"


class HotelMixin:
    def chapter(self, text):
        bar=Line([-6.6,2.85,0],[6.6,2.85,0],color="#2B4154",stroke_width=1)
        sub=label(text,[0,2.45,0],GOLD,21)
        self.add(bar,sub)
        return sub

    def badge(self, text, point, color=GOLD, size=20):
        txt=label(text,point,color,size)
        back=RoundedRectangle(width=txt.width+.5,height=txt.height+.35,corner_radius=.1,
                              color=color,fill_color=PANEL,fill_opacity=.95).move_to(txt)
        return VGroup(back,txt)

    def hud(self, title_text, number, fixed=False):
        self.title = Text(title_text.upper(), font="Arial", font_size=26, color=ICE)
        self.title.to_corner(UL, buff=.45)
        self.lecture = VGroup()
        if fixed:
            self.add_fixed_in_frame_mobjects(self.title)
        else:
            self.add(self.title)


class HotelScene(HotelMixin, TeachingScene):
    pass


class Hotel3DScene(HotelMixin, Teaching3DScene):
    pass


def room_front(n, point, color=CYAN, width=1.35, height=1.45):
    shell=RoundedRectangle(width=width,height=height,corner_radius=.07,color="#426075",
                           fill_color="#101E2C",fill_opacity=1,stroke_width=2)
    door=Rectangle(width=width*.68,height=height*.66,color=color,fill_color="#21384B",
                   fill_opacity=.8,stroke_width=1.4).shift(DOWN*height*.08)
    lamp=Line(LEFT*width*.25,RIGHT*width*.25,color=color,stroke_width=5).shift(UP*height*.27)
    knob=Dot([width*.22,-height*.16,0],radius=.025,color=GOLD)
    plaque=label(str(n),[0,height*.41,0],ICE,19 if width>1 else 15)
    return VGroup(shell,door,lamp,knob,plaque).move_to(point)


def bus_shape(point, color=GOLD):
    body=RoundedRectangle(width=2.7,height=.9,corner_radius=.15,color=color,
                          fill_color="#5A4223",fill_opacity=1,stroke_width=2)
    windows=VGroup(*[Rectangle(width=.38,height=.29,color=ICE,fill_color="#182F43",fill_opacity=1)
                     .move_to([-.95+i*.49,.16,0]) for i in range(4)])
    wheels=VGroup(*[Circle(radius=.17,color=DIM,fill_color=INK,fill_opacity=1).move_to([x,-.43,0])
                    for x in [-.85,.85]])
    light=Dot([1.32,-.15,0],radius=.05,color=GOLD)
    return VGroup(body,windows,wheels,light).move_to(point)


def travel_path(a,b,arc=.45):
    return ArcBetweenPoints(np.array(a),np.array(b),angle=arc)


def label(text, point, color=ICE, size=18):
    return Text(text, font="Arial", font_size=size, color=color).move_to(point)


def guest(scale=.25, color=ALARM):
    body = RoundedRectangle(width=.26, height=.30, corner_radius=.07, color=color, fill_color=color, fill_opacity=1)
    head = Circle(radius=.09, color=ICE, fill_color=ICE, fill_opacity=1).shift(UP*.2)
    bag = Rectangle(width=.13, height=.18, color=GOLD, fill_color=GOLD, fill_opacity=1).next_to(body, RIGHT, buff=.04)
    feet = VGroup(Line([-.07,-.14,0],[-.09,-.24,0],color=color,stroke_width=3),
                  Line([.07,-.14,0],[.10,-.24,0],color=color,stroke_width=3))
    return VGroup(body, head, bag, feet).scale(scale)


def room(number, center, occupied=True, accent=CYAN, width=.88, height=.62, font_size=14):
    box = Rectangle(width=width, height=height, color=accent if occupied else DIM,
                    fill_color="#14283A" if occupied else "#08121D", fill_opacity=1, stroke_width=2)
    box.move_to(center)
    number_text = label(str(number), np.array(center)+UP*height*.27, ICE if occupied else DIM, font_size)
    people = guest(.45, ALARM if occupied else DIM).move_to(np.array(center)+DOWN*height*.19) if occupied else VGroup()
    return VGroup(box, number_text, people)


def hotel_row(numbers, y=0, start_x=-4.2, spacing=1.0, occupied=True, accent=CYAN):
    return VGroup(*[room(n, [start_x+i*spacing, y, 0], occupied=occupied, accent=accent) for i, n in enumerate(numbers)])


def corridor_background():
    lines = VGroup(*[
        Line([-6.8, y, 0], [6.8, y*.38, 0], color="#1D3444", stroke_width=1)
        for y in np.linspace(-3.6, 3.6, 9)
    ])
    ceiling = VGroup(*[Line([x, 3.2, 0], [x*.38, .5, 0], color="#173042", stroke_width=1)
                       for x in np.linspace(-6, 6, 9)])
    return VGroup(lines, ceiling)


def infinity_sign(point, scale=1.0, color=GOLD):
    return MathTex(r"\infty", color=color, font_size=46).scale(scale).move_to(point)


def arrows_between(srcs, dsts, color=GOLD, buff=.08):
    return VGroup(*[Arrow(np.array(a)+UP*buff, np.array(b)+DOWN*buff, color=color,
                          stroke_width=2, max_tip_length_to_length_ratio=.18)
                    for a, b in zip(srcs, dsts)])
