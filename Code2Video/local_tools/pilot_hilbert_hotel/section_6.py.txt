from hotel_scene_base import *

class Section6Scene(Hotel3DScene):
    def construct(self):
        self.setup_3d_layout("WELCOME TO HILBERT'S HOTEL", [], phi=70*DEGREES,theta=-75*DEGREES,number=6)
        self.set_camera_orientation(zoom=.93)
        # The building is illustration. Room numbers are the authoritative model.
        ground=Prism(dimensions=[7.6,4.2,.18],fill_color="#152635",fill_opacity=1,stroke_width=.3).shift(DOWN*.25+IN*2.0)
        body=Prism(dimensions=[5.5,2.6,4.4],fill_color="#152B40",fill_opacity=.95,stroke_color=DIM,stroke_width=.5).shift(OUT*.25)
        roof=Prism(dimensions=[5.8,2.9,.18],fill_color=GOLD,fill_opacity=1,stroke_width=.4).shift(OUT*2.53)
        ledges=VGroup(*[Prism(dimensions=[5.65,2.75,.06],fill_color="#425A70",fill_opacity=1,stroke_width=0).shift(OUT*z) for z in [-1.0,.15,1.3]])
        windows=VGroup()
        for row in range(3):
            for col in range(5):
                w=Rectangle(width=.6,height=.72,fill_color="#66502A",fill_opacity=.75,color=GOLD,stroke_width=1.3)
                w.rotate(PI/2,axis=RIGHT).move_to([-2.1+col*1.05,-1.315,1.85-row*1.08])
                windows.add(w)
        lobby=Rectangle(width=.75,height=.92,color=GREEN,fill_color="#133E31",fill_opacity=1,stroke_width=2).rotate(PI/2,axis=RIGHT).move_to([0,-1.335,-1.23])
        canopy=Prism(dimensions=[1.5,.85,.12],fill_color=GOLD,fill_opacity=1,stroke_width=.5).move_to([0,-1.7,-.69])
        pillars=VGroup(*[Cylinder(radius=.045,height=.95,fill_color=GOLD,fill_opacity=1,stroke_width=0).move_to([x,-1.95,-1.22]) for x in [-.63,.63]])
        model=VGroup(ground,body,ledges,roof)
        self.play(FadeIn(model,shift=OUT*.2),run_time=2.5)
        self.play(LaggedStart(*[FadeIn(w) for w in windows],lag_ratio=.07),run_time=2.5)
        self.play(FadeIn(lobby),FadeIn(canopy),FadeIn(pillars),run_time=1.5)
        self.move_camera(theta=-61*DEGREES,phi=64*DEGREES,run_time=4)
        rule=MathTex(r"n\longmapsto n+1",color=GOLD,font_size=37).move_to([4.1,1.9,0])
        meaning=label("COUNTABLE INFINITY",[4.1,.95,0],ICE,21)
        self.add_fixed_in_frame_mobjects(rule,meaning)
        self.play(Write(rule),FadeIn(meaning),run_time=1.5)
        # Light wave reacts to reassignment; a luggage token moves into the lobby.
        self.play(LaggedStart(*[w.animate.set_fill("#EBC96B",opacity=.95) for w in windows],lag_ratio=.06),run_time=2.5)
        token=Prism(dimensions=[.19,.12,.27],fill_color=PINK,fill_opacity=1,stroke_color=PINK,stroke_width=.5).move_to([-1.5,-2.7,-1.69])
        self.play(FadeIn(token),run_time=.5)
        self.play(token.animate.move_to([0,-1.6,-1.69]),lobby.animate.set_fill(GREEN,opacity=.7),run_time=2)
        self.play(FadeOut(token),run_time=.5)
        conclusion=label("A RULE MAKES ROOM.",[0,-3.2,0],GREEN,30)
        self.add_fixed_in_frame_mobjects(conclusion)
        self.play(Write(conclusion),run_time=1.5)
        self.move_camera(theta=-70*DEGREES,zoom=.83,run_time=3)
        self.finish_slot()
