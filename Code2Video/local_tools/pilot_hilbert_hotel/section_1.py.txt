from hotel_scene_base import *

class Section1Scene(HotelScene):
    def construct(self):
        self.setup_layout("HILBERT'S INFINITE HOTEL", [], number=1)
        self.chapter("Every door occupied. The corridor never ends.")
        floor=Polygon([-6.8,-2.65,0],[6.8,-2.65,0],[2.2,1.8,0],[-2.2,1.8,0],color="#1A3247",fill_color="#0A1420",fill_opacity=1)
        seams=VGroup(*[Line([x,-2.65,0],[x*.18,1.8,0],color="#21354A",stroke_width=1) for x in np.linspace(-6.6,6.6,9)])
        doors=VGroup(*[room_front(i+1,[-5.4+i*1.8,.25,0],color=GOLD) for i in range(6)])
        far=VGroup(*[room_front(i+7,[3.6-i*.38,1.55+i*.09,0],width=.45,height=.52).set_opacity(.7-i*.1) for i in range(5)])
        self.play(FadeIn(floor),Create(seams),run_time=1.5)
        self.play(LaggedStart(*[FadeIn(d,shift=UP*.15) for d in doors],lag_ratio=.1),run_time=2)
        self.play(FadeIn(far),run_time=1.5)
        people=VGroup(*[guest(1.05,BLUE).move_to(d.get_center()+DOWN*.2) for d in doors])
        occupied=self.badge("ALL ROOMS OCCUPIED",[0,-1.5,0],ALARM,24)
        self.play(LaggedStart(*[FadeIn(p) for p in people],lag_ratio=.13),FadeIn(occupied),run_time=2)
        inf=infinity_sign([5.65,1.55,0],.8)
        self.play(Write(inf),run_time=1)
        self.wait(1.5)
        newcomer=guest(1.8,PINK).move_to([-7.3,-2.3,0])
        desk=RoundedRectangle(width=2.15,height=.5,corner_radius=.08,color=GOLD,fill_color="#45351D",fill_opacity=1).move_to([-4.8,-2.65,0])
        desk_name=label("RECEPTION",[-4.8,-3.13,0],GOLD,17)
        self.play(FadeIn(desk),FadeIn(desk_name),newcomer.animate.move_to([-3.1,-2.25,0]),run_time=2.2)
        bell=VGroup(Arc(radius=.14,angle=PI,color=GOLD).shift(UP*.04),Line(LEFT*.2,RIGHT*.2,color=GOLD)).move_to([-5.25,-2.34,0])
        self.play(FadeIn(bell),run_time=.6)
        self.wait(2)
        self.play(Indicate(occupied,color=ALARM),run_time=1)
        question=self.badge("FULL ... BUT ONE MORE GUEST?",[2.45,-2.75,0],ICE,23)
        self.play(FadeIn(question,shift=UP*.15),run_time=1)
        self.play(people.animate.set_opacity(.6),newcomer.animate.shift(RIGHT*.2),run_time=2)
        self.finish_slot()
