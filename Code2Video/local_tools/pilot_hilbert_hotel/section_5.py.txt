from hotel_scene_base import *

class Section5Scene(HotelScene):
    def construct(self):
        self.setup_layout("SIZE IS A MATCHING RULE", [], number=5)
        self.chapter("Countable infinity is measured by one-to-one pairing.")
        left=RoundedRectangle(width=5.75,height=4.35,corner_radius=.12,color=ALARM,fill_color="#21141A",fill_opacity=1).move_to([-3.2,-.2,0])
        right=RoundedRectangle(width=5.75,height=4.35,corner_radius=.12,color=BLUE,fill_color="#101E30",fill_opacity=1).move_to([3.2,-.2,0])
        self.play(FadeIn(left),FadeIn(right),run_time=1.5)
        headings=VGroup(label("FINITE HOTEL",[-3.2,1.55,0],ALARM,23),label("COUNTABLY INFINITE",[3.2,1.55,0],BLUE,23))
        finite=VGroup(*[room_front(i+1,[-5.1+i*1.2,.5,0],width=.98,height=1.0,color=ALARM) for i in range(4)])
        people=VGroup(*[guest(.85,ALARM).move_to(d.get_center()+DOWN*.17) for d in finite])
        newcomer=guest(1.0,PINK).move_to([-3.2,-1.1,0])
        self.play(FadeIn(headings),FadeIn(finite),FadeIn(people),FadeIn(newcomer),run_time=2)
        block=label("NO NEXT ROOM",[-3.2,-1.85,0],ALARM,20)
        self.play(FadeIn(block),newcomer.animate.shift(UP*.2),run_time=1)
        self.wait(1)
        old_labels=VGroup(*[label(str(i+1),[1.35+i*1.05,.55,0],BLUE,27) for i in range(4)])
        new_labels=VGroup(*[label(str(i+2),[1.35+i*1.05,-.85,0],GREEN,27) for i in range(4)])
        ellipses=VGroup(label("...",[5.65,.55,0],BLUE,27),label("...",[5.65,-.85,0],GREEN,27))
        self.play(FadeIn(old_labels),FadeIn(new_labels),FadeIn(ellipses),run_time=1.5)
        matches=VGroup(*[Arrow([1.35+i*1.05,.25,0],[1.35+i*1.05,-.55,0],color=GOLD,buff=0,stroke_width=2) for i in range(4)])
        self.play(LaggedStart(*[Create(a) for a in matches],lag_ratio=.18),run_time=2)
        equation=MathTex(r"\aleph_0+1=\aleph_0",color=GREEN,font_size=35).move_to([3.2,-1.85,0])
        self.play(Write(equation),run_time=1.5)
        vacancy=self.badge("NEW GUEST  →  ROOM 1",[0,-3.05,0],PINK,23)
        self.play(FadeIn(vacancy),run_time=1)
        self.play(Indicate(matches,color=GOLD),run_time=1)
        self.wait(2)
        self.play(left.animate.set_opacity(.45),equation.animate.scale(1.06),run_time=1.5)
        self.finish_slot()
