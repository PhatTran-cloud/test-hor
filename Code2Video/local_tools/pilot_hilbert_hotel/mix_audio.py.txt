"""Hotel-specific procedural Foley, timed to the rendered actions (seconds)."""
import json
import subprocess
import sys
from pathlib import Path

import numpy as np
from scipy.io import wavfile
from scipy.ndimage import uniform_filter1d

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT / "src"))
from output_paths import get_output_root
OUT = get_output_root() / "hilberts_infinite_hotel"
RATE = 48000


def main():
    clips = json.loads((OUT / "audio_timeline.json").read_text(encoding="utf-8"))
    count = RATE * 180
    voice = np.zeros(count, dtype=np.float32)
    for clip in clips:
        raw = subprocess.run(["ffmpeg", "-v", "error", "-i", clip["file"], "-f", "f32le",
                              "-ac", "1", "-ar", str(RATE), "pipe:1"], check=True, capture_output=True).stdout
        samples = np.frombuffer(raw, dtype="<f4")
        active = samples[np.abs(samples) > .025]
        rms = np.sqrt(np.mean(active**2)) if len(active) else .001
        gain = min(1.5, .115 / max(rms, .001))
        start = round(clip["start"] * RATE)
        voice[start:start+len(samples)] += samples * gain
    envelope = np.sqrt(np.maximum(uniform_filter1d(voice*voice, size=4800), 0))
    duck = 1 - .45*np.clip(envelope/.03, 0, 1)
    time = np.arange(count, dtype=np.float32)/RATE
    bed = (.004*np.sin(2*np.pi*130.81*time)+.003*np.sin(2*np.pi*196*time))
    bed *= np.clip(time/4, 0, 1)*np.clip((180-time)/5, 0, 1)*duck
    mix = np.column_stack([voice+bed, voice+bed])
    rng = np.random.default_rng(20261009)
    ledger = []

    def cue(at, kind, duration=.4, gain=.055, pan=0, trigger=""):
        t = np.arange(round(duration*RATE), dtype=np.float32)/RATE
        phase = np.clip(t/duration, 0, 1)
        noise = rng.normal(0, 1, len(t)).astype(np.float32)
        if kind == "bell":
            sound = (np.sin(2*np.pi*880*t)+.5*np.sin(2*np.pi*1320*t)+.25*np.sin(2*np.pi*1760*t))*np.exp(-t*5)
        elif kind == "footstep":
            sound = (.7*np.sin(2*np.pi*125*t)+.25*noise)*np.exp(-t*38)
        elif kind == "door":
            sound = (.55*np.sin(2*np.pi*(90*t+45*t*t))+noise*.12)*np.sin(np.pi*phase)*np.exp(-t*3)
        elif kind == "bus":
            sound = (.45*np.sin(2*np.pi*65*t)+.2*np.sin(2*np.pi*97*t)+noise*.14)*np.sin(np.pi*phase)
        elif kind == "whoosh":
            smooth_noise = uniform_filter1d(noise, size=12)
            sound = smooth_noise*np.sin(np.pi*phase)*2.3
        elif kind == "key":
            sound = (np.sin(2*np.pi*2100*t)+.35*np.sin(2*np.pi*3200*t))*np.exp(-t*32)
        else:
            sound = (np.sin(2*np.pi*523.25*t)+.45*np.sin(2*np.pi*783.99*t))*np.exp(-t*7)
        sound *= np.minimum(t/.008, 1)*gain
        start = round(at*RATE)
        end = min(count, start+len(sound))
        sound = sound[:end-start]*duck[start:end]
        mix[start:end,0] += sound*(1-pan*.4)
        mix[start:end,1] += sound*(1+pan*.4)
        ledger.append({"time": at, "type": kind, "duration": duration, "gain": gain, "trigger": trigger})

    cue(1.5, "whoosh", 2, .035, trigger="corridor doors reveal")
    cue(9.7, "footstep", .14, trigger="new guest arrival")
    cue(10.3, "footstep", .14)
    cue(11.9, "bell", .8, .06, -.4, "reception bell appears")
    cue(15.5, "key", .2, .05, trigger="question reveal")
    for at in [36, 36.6, 37.2, 37.8, 38.4]:
        cue(at, "footstep", .13, .048, trigger="all guests shift one address")
    cue(40, "door", .7, .055, -.5, "room one opens")
    cue(43.2, "key", .18, .055, -.5, "new guest enters room one")
    cue(61.5, "bus", 1.5, .08, .5, "bus enters")
    cue(65.6, "whoosh", 3, .045, trigger="old guests travel to even rooms")
    cue(70.5, "reveal", .5, .06, trigger="odd rooms light up")
    for at in [71.9, 72.5, 73.1, 73.7, 74.3]:
        cue(at, "footstep", .12, .045, trigger="bus passengers enter odd rooms")
    for at in [98.15, 100.50, 102.85, 105.20]:
        cue(at, "reveal", .35, .055, trigger="bus-seat pair reaches its unique room")
    cue(107.3, "bell", .6, .04, trigger="unique-factorization statement")
    cue(124.5, "door", .35, .06, -.5, "finite hotel has no next room")
    for at in [127, 127.4, 127.8, 128.2]:
        cue(at, "key", .16, .04, .4, "one-to-one pairing")
    cue(130, "reveal", .6, .065, trigger="cardinality equation")
    cue(152.5, "whoosh", 2.5, .04, trigger="hotel windows reveal")
    cue(156.5, "whoosh", 4, .025, trigger="camera orbits hotel")
    cue(162, "key", .25, .05, trigger="light wave reassignment")
    cue(165.5, "footstep", .14, .05, trigger="luggage token approaches lobby")
    cue(167, "door", .5, .06, trigger="lobby welcomes guest")
    cue(168, "bell", 1.1, .07, trigger="a rule makes room")
    for at in [29.0, 59.0, 89.0, 119.0, 149.0]:
        cue(at, "whoosh", .65, .035, trigger="scene handoff")
    if not np.isfinite(mix).all():
        raise ValueError("Non-finite audio samples")
    peak = float(np.max(np.abs(mix)))
    mix *= min(1, .94/max(peak, .001))
    wavfile.write(OUT / "soundtrack.wav", RATE, (np.clip(mix,-1,1)*32767).astype(np.int16))
    (OUT / "sound_design.json").write_text(json.dumps({"duration":180,"rate":RATE,
        "provenance":"Original deterministic hotel Foley and music; local Kokoro narration",
        "events":ledger,"ducking":"speech RMS envelope","peak_before_gain":peak},indent=2),encoding="utf-8")
    print(f"Hotel mix: {len(ledger)} action-linked SFX, 180s stereo")


if __name__ == "__main__":
    main()
