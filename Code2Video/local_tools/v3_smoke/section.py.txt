from manim import *


class Section1Scene(Scene):
    def construct(self):
        self.camera.background_color = "#05070D"
        line = Line(LEFT * 3, RIGHT * 3, color="#456878")
        circle = Circle(radius=.7, color="#9BBCCD").shift(LEFT * 3)
        target = circle.copy().shift(RIGHT * 6)
        dot = Dot(LEFT * 3, color="#64D9E8")
        self.add(line, circle, target, dot)
        self.play(dot.animate.shift(RIGHT * 6), run_time=1.5, rate_func=smooth)
        self.wait(.5)
