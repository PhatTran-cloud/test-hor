"""Run the v3 fixture with measured Kokoro timing and one explicitly selected SFX.

This tests integration, not final listening, scientific or artistic acceptance.
"""
import json
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT / "src"))
from kokoro_tts import prepare_scene_speech
from renderers import probe_video
from sfx_assets import prepare_cues
from output_paths import get_output_root


def main():
    fixture = Path(__file__).resolve().parent
    evidence = get_output_root() / "v3_verification"
    evidence.mkdir(parents=True, exist_ok=True)
    plan = json.loads((fixture / "plan.json").read_text())
    section = plan["sections"][0]
    timing = prepare_scene_speech(section["narration"], evidence / "voice.wav")
    section["shots"][0]["source"] = str(fixture / "section.py")
    for shot in section["shots"][1:]:
        shot["render_spec"]["project_dir"] = str(fixture / "browser")
    remaining = timing["frames"] - 240
    if remaining < 30:
        raise ValueError("Fixture narration is too short for its authored first two shots")
    section["shots"][2]["duration_frames"] = remaining
    # Chosen from the recorded candidate set by restrained low-frequency accent
    # intent. Metadata fit is not a listening approval.
    section["sfx_cues"] = prepare_cues([{"sound_id": 669468, "trigger": "Marker reaches destination",
        "purpose": "Restrained schematic accent, not a literal natural sound", "role": "event",
        "relative_time": 3, "duration": .65, "gain_db": -21,
        "fade_in": .012, "fade_out": .08, "selection_status": "METADATA_SELECTED_LISTENING_UNVERIFIED"}], fixture)
    plan["sound_design"] = "adaptive"
    generated = evidence / "plan.json"
    generated.write_text(json.dumps(plan, indent=2), encoding="utf-8")
    result = subprocess.run([sys.executable, str(ROOT / "local_tools/render_local.py"), str(generated)],
        cwd=ROOT, capture_output=True, encoding="utf-8", errors="replace", timeout=900)
    (evidence / "render.log").write_text(result.stdout + "\n" + result.stderr, encoding="utf-8")
    if result.returncode:
        raise RuntimeError(f"Inspect {evidence / 'render.log'}")
    marker = result.stdout.rfind('{\n  "video"')
    if marker < 0:
        raise ValueError("Local renderer did not report final video")
    video = json.loads(result.stdout[marker:])["video"]
    technical = probe_video(video)
    if int(technical["nb_frames"]) != timing["frames"]:
        raise ValueError("Final video must preserve the measured speech/shot frame total")
    subprocess.run(["ffmpeg", "-v", "error", "-i", video, "-f", "null", "-"], check=True)
    subprocess.run(["ffmpeg", "-y", "-i", video, "-vf",
        "fps=2,scale=480:270,tile=4x3", "-frames:v", "1", str(evidence / "contact.jpg")],
        check=True, capture_output=True)
    report = {"video": video, "speech_frames": timing["frames"], "technical": technical,
        "decode_status": "PASS", "visual_status": "UNVERIFIED", "scientific_status": "UNVERIFIED",
        "sfx_listening_status": "UNVERIFIED", "llm_api_calls": 0,
        "sound_source": section["sfx_cues"][0]["provenance"]}
    (evidence / "report.json").write_text(json.dumps(report, indent=2), encoding="utf-8")
    print(json.dumps(report, indent=2))


if __name__ == "__main__":
    main()
