from elnino_scene_base import *

class Section6Scene(ClimateScene):
    def construct(self):
        self.header('TWO PHASES. ONE COUPLED SYSTEM.', 'ENSO / EL NINO - SOUTHERN OSCILLATION')
        warm_box=RoundedRectangle(width=5.9,height=3.6,corner_radius=.14,stroke_color=HOT,stroke_width=1.3,fill_color='#22272A',fill_opacity=1).move_to([-3.15,.45,0])
        cool_box=warm_box.copy().move_to([3.15,.45,0]).set_stroke(COLD).set_fill('#102C3C',opacity=1)
        self.play(FadeIn(warm_box),FadeIn(cool_box),run_time=1)
        warm_title=label('EL NINO',[-3.15,1.7,0],HOT,31)
        cool_title=label('LA NINA',[3.15,1.7,0],COLD,31)
        self.play(FadeIn(warm_title),run_time=.7)
        # Both wind arrows point west; El Nino arrows are shorter, never mislabeled ocean flow.
        weak=wind_set(left=-5.3,right=-1.4,y=.95,direction=-1,color=GOLD,count=5)
        for a in weak: a.scale(.4,about_point=a.get_start())
        warm=RoundedRectangle(width=4.9,height=.48,corner_radius=.14,fill_color=HOT,fill_opacity=.85,stroke_width=0).move_to([-3.15,.15,0])
        self.play(FadeIn(weak),FadeIn(warm),run_time=1)
        flow=flow_dots(Line([-5.45,.15,0],[-.9,.15,0]),ICE,10,.055)
        self.add(flow)
        self.until(7)
        self.play(FadeIn(cool_title),run_time=.5)
        strong=wind_set(left=1.6,right=5.5,y=.95,direction=-1,color=GOLD,count=5)
        cool=RoundedRectangle(width=3.9,height=.48,corner_radius=.14,fill_color=COLD,fill_opacity=.85,stroke_width=0).move_to([3.8,.15,0])
        west=RoundedRectangle(width=.75,height=.48,corner_radius=.1,fill_color=WARM,fill_opacity=.85,stroke_width=0).move_to([1.2,.15,0])
        self.play(FadeIn(strong),FadeIn(cool),FadeIn(west),run_time=1)
        cold_flow=flow_dots(Line([5.5,.15,0],[1.7,.15,0]),ICE,10,.11)
        self.add(cold_flow)
        details=VGroup(label('Warmer central / eastern Pacific',[-3.15,-.75,0],MUTED,18),label('Cooler central / eastern Pacific',[3.15,-.75,0],MUTED,18))
        self.play(FadeIn(details),run_time=1)
        self.until(15)
        neutral=label('NEUTRAL: BETWEEN THE WARM AND COOL PHASES',[0,-1.85,0],ICE,20)
        self.play(FadeIn(neutral),run_time=.7)
        self.until(20)
        period=label('IRREGULAR RETURN: OFTEN 2-7 YEARS',[0,-2.55,0],MUTED,21)
        self.play(FadeIn(period),run_time=.7)
        self.until(26)
        self.footer('Wind moves ocean heat. Ocean heat reorganizes weather.',GREEN)
        self.finish_slot()
