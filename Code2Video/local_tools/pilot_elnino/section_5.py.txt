from elnino_scene_base import *

class Section5Scene(Climate3DScene):
    def construct(self):
        self.set_camera_orientation(phi=82*DEGREES,theta=-150*DEGREES,zoom=.92)
        self.header('ONE OCEAN. A GLOBAL RESPONSE.', 'EL NINO / ATMOSPHERIC TELECONNECTIONS')
        globe=geo_globe(2.25,self.camera)
        self.add(globe)
        self.begin_ambient_camera_rotation(rate=.0035)
        patch=Surface(lambda u,v:geo_point(v,u,2.34),u_range=[-5,5],v_range=[-170,-90],resolution=(4,20),fill_color=HOT,fill_opacity=.95,stroke_width=0,checkerboard_colors=False)
        self.play(FadeIn(patch),run_time=1.2)
        self.until(5)
        # These arcs are schematic atmospheric influence paths, not ocean currents.
        wave=ParametricFunction(lambda t:geo_point(-175+90*t,18+12*np.sin(2*PI*t),2.48),t_range=[0,1,.01],color=RAIN,stroke_width=2)
        self.play(Create(wave),run_time=2.5)
        moving=Dot3D(radius=.055,color=RAIN).move_to(wave.get_start())
        self.play(MoveAlongPath(moving,wave),run_time=2.6,rate_func=linear)
        self.until(11)
        peru=Dot3D(geo_point(-80,-10,2.30),color=RAIN,radius=.06)
        self.add(peru)
        wet=label('COASTAL PERU',[4.65,1.65,0],RAIN,21)
        wet_small=label('Heavier rain risk',[4.65,1.15,0],MUTED,19)
        self.add_fixed_in_frame_mobjects(wet,wet_small)
        self.until(15)
        dry=label('INDONESIA / AUSTRALIA',[-4.85,1.65,0],GOLD,18)
        dry_small=label('Greater dry risk',[-4.85,1.15,0],MUTED,19)
        self.add_fixed_in_frame_mobjects(dry,dry_small)
        self.until(20)
        self.remove(moving)
        self.footer('Changes in probability. Not identical weather everywhere.',ICE)
        self.until(25)
        caveat=label('Season + event strength + other climate patterns',[0,-2.93,0],MUTED,18)
        self.add_fixed_in_frame_mobjects(caveat)
        self.finish_slot()
