def label(text, point, color=ICE, size=22):
    return Text(text, font='Arial', font_size=size, color=color).move_to(point)


def clip_polygon(points, xmin, xmax, ymin, ymax):
    result = [np.array(p[:2], dtype=float) for p in points]
    for axis, limit, lower in [(0,xmin,True),(0,xmax,False),(1,ymin,True),(1,ymax,False)]:
        if not result: break
        new=[]
        a=result[-1]; inside_a=a[axis]>=limit if lower else a[axis]<=limit
        for b in result:
            inside_b=b[axis]>=limit if lower else b[axis]<=limit
            if inside_a != inside_b:
                t=(limit-a[axis])/(b[axis]-a[axis])
                new.append(a+(b-a)*t)
            if inside_b: new.append(b)
            a=b; inside_a=inside_b
        result=new
    return result


def land_rings():
    data=json.loads(DATA.read_text(encoding='utf-8'))
    for f in data['features']:
        g=f['geometry']
        for p in ([g['coordinates']] if g['type']=='Polygon' else g['coordinates']):
            yield p[0]


class PacificMap(VGroup):
    def __init__(self, width=12.0, center=DOWN*.15, **kwargs):
        super().__init__(**kwargs)
        self.width_units=width; self.map_center=np.array(center)
        self.scale_geo=width/200
        self.bbox=(100,300,-44,46)
        sea=Rectangle(width=width,height=90*self.scale_geo,stroke_width=0,
                      fill_color=OCEAN,fill_opacity=1).move_to(self.xy(200,1))
        self.grid=VGroup()
        for lon in range(100,301,20):
            self.grid.add(Line(self.xy(lon,-44),self.xy(lon,46),color=GRID,stroke_width=.7))
        for lat in range(-40,41,20):
            self.grid.add(Line(self.xy(100,lat),self.xy(300,lat),color=GRID,stroke_width=.7))
        self.land=VGroup()
        for ring in land_rings():
            for offset in (0,360):
                pts=clip_polygon([(p[0]+offset,p[1]) for p in ring],*self.bbox)
                if len(pts)>2:
                    self.land.add(Polygon(*[self.xy(*p) for p in pts],stroke_color=COAST,
                                           stroke_width=.65,fill_color=LAND,fill_opacity=1))
        equator=DashedLine(self.xy(100,0),self.xy(300,0),color=MUTED,
                           stroke_width=.8,dash_length=.06)
        frame=Rectangle(width=width,height=90*self.scale_geo,color=GRID,stroke_width=1.3).move_to(self.xy(200,1))
        self.add(sea,self.grid,equator,self.land,frame)

    def xy(self, lon, lat):
        if lon<0: lon+=360
        return self.map_center+np.array([(lon-200)*self.scale_geo,lat*self.scale_geo,0])

    def region_labels(self):
        return VGroup(label('INDONESIA',self.xy(123,13),ICE,17),
                      label('AUSTRALIA',self.xy(135,-30),INK,17),
                      label('S. AMERICA',self.xy(286,-20),INK,15),
                      label('PACIFIC OCEAN',self.xy(212,-27),MUTED,19))


def heat_ribbon(m, west, east, color=HOT, lat=0):
    mid=(west+east)/2; width=(east-west)*m.scale_geo
    layers=VGroup()
    for height, opacity in [(1.15,.09),(.85,.12),(.6,.22),(.34,.6)]:
        layers.add(RoundedRectangle(width=width,height=height,corner_radius=min(.16,height/2),
                                    stroke_width=0,fill_color=color,fill_opacity=opacity).move_to(m.xy(mid,lat)))
    return layers


def flow_dots(path, color=COLD, count=12, speed=.09, radius=.032):
    dots=VGroup()
    for i in range(count):
        dot=Dot(path.point_from_proportion(i/count),radius=radius,color=color)
        dot.phase=i/count
        def update(d,dt,p=path):
            d.phase=(d.phase+speed*dt)%1
            d.move_to(p.point_from_proportion(d.phase))
        dot.add_updater(update); dots.add(dot)
    return dots


def wind_set(left=-3.6,right=3.6,y=1.3,direction=-1,color=GOLD,count=6):
    return VGroup(*[Arrow([x,y,0],[x+.7*direction,y,0],buff=0,stroke_width=2.8,color=color,
                           max_tip_length_to_length_ratio=.19) for x in np.linspace(left,right,count)])


def cloud(point, scale=1):
    g=VGroup(*[Circle(radius=r,stroke_width=0,fill_color=RAIN,fill_opacity=.96).shift(RIGHT*x+UP*y)
               for x,y,r in [(-.36,0,.2),(-.14,.11,.25),(.14,.13,.27),(.39,.02,.2)]],
              RoundedRectangle(width=.96,height=.28,corner_radius=.1,stroke_width=0,
                               fill_color=RAIN,fill_opacity=.96).shift(DOWN*.08))
    return g.scale(scale).move_to(point)


def falling_rain(point, width=.8, count=14):
    x,y,_=point; group=VGroup()
    for i in range(count):
        phase=i/count
        xx=x-width/2+width*((i*7%count)/max(1,count-1))
        streak=Line([xx,y-phase*.8,0],[xx-.015,y-phase*.8-.095,0],color=RAIN,stroke_width=1.3)
        streak.phase=phase
        def update(s,dt,xx=xx):
            s.phase=(s.phase+dt*1.1)%1
            yy=y-.8*s.phase
            s.put_start_and_end_on([xx,yy,0],[xx-.015,yy-.095,0])
            s.set_opacity(np.sin(PI*s.phase))
        streak.add_updater(update); group.add(streak)
    return group


class OceanSlice(VGroup):
    def __init__(self, state=0, **kwargs):
        super().__init__(**kwargs)
        self.state=ValueTracker(state)
        bottom=Rectangle(width=11.6,height=3.4,stroke_width=0,fill_color=DEEP,fill_opacity=1).move_to([0,-1.4,0])
        self.add(bottom)
        self.warm=VMobject(stroke_width=0,fill_color=WARM,fill_opacity=.94)
        self.boundary=VMobject(stroke_color=COLD,stroke_width=3.4)
        self.add(self.warm,self.boundary)
        self.warm.add_updater(lambda _: self.update_geometry())
        self.update_geometry()
        self.surface=Line([-5.8,.3,0],[5.8,.3,0],color=ICE,stroke_width=1.4)
        self.add(self.surface)

    def depth(self,x):
        normal=-1.7+.16*x
        return (1-self.state.get_value())*normal+self.state.get_value()*(-1.5)

    def update_geometry(self):
        xs=np.linspace(-5.8,5.8,60)
        curve=[[x,self.depth(x),0] for x in xs]
        self.boundary.set_points_smoothly(curve)
        self.warm.set_points_as_corners([[-5.8,.3,0],[5.8,.3,0],*reversed(curve),[-5.8,.3,0]])
        self.warm.set_fill(interpolate_color(ManimColor(WARM),ManimColor(HOT),self.state.get_value()),opacity=.94)


class ClimateMixin:
    def header(self, title, kicker):
        self.camera.background_color=BG
        self.native_duration=30.0
        timing_path=Path(__file__).parent / 'speech_timing.json'
        if timing_path.exists():
            sid='section_'+re.search(r'Section(\d+)Scene',type(self).__name__).group(1)
            self.native_duration=json.loads(timing_path.read_text(encoding='utf-8'))[sid]['duration']
        title_obj=label(title,[-6.35,3.07,0],ICE,38).align_to([-6.35,0,0],LEFT)
        kicker_obj=label(kicker,[-6.35,3.62,0],MUTED,15).align_to([-6.35,0,0],LEFT)
        if isinstance(self,ThreeDScene): self.add_fixed_in_frame_mobjects(title_obj,kicker_obj)
        else: self.add(title_obj,kicker_obj)

    def footer(self, text, color=ICE):
        note=label(text,[0,-3.43,0],color,25)
        if isinstance(self,ThreeDScene): self.add_fixed_in_frame_mobjects(note)
        else: self.add(note)
        return note

    def until(self,t):
        remaining=t*self.native_duration/30-self.renderer.time
        if remaining>0: self.wait(remaining)

    def finish_slot(self):
        self.until(30)
        if abs(self.renderer.time-self.native_duration)>.04: raise ValueError(f'Bad scene duration: {self.renderer.time}')


class ClimateScene(ClimateMixin,Scene): pass
class Climate3DScene(ClimateMixin,ThreeDScene): pass


def geo_point(lon,lat,radius=2.3):
    lon,lat=np.radians([lon,lat])
    return radius*np.array([np.cos(lat)*np.cos(lon),np.cos(lat)*np.sin(lon),np.sin(lat)])


def geo_globe(radius=2.3, camera=None):
    mask=Image.new('L',(1440,720)); draw=ImageDraw.Draw(mask)
    for ring in land_rings():
        draw.polygon([((p[0]+180)*4,(90-p[1])*4) for p in ring],fill=255)
    sphere=Sphere(radius=radius,resolution=(32,64),checkerboard_colors=False,
                  stroke_width=0,fill_opacity=1,fill_color=OCEAN)
    for patch in sphere:
        c=patch.get_center(); lat=np.degrees(np.arctan2(c[2],np.hypot(c[0],c[1]))); lon=np.degrees(np.arctan2(c[1],c[0]))
        px=min(1439,max(0,int((lon+180)*4))); py=min(719,max(0,int((90-lat)*4)))
        patch.set_fill(LAND if mask.getpixel((px,py)) else OCEAN,opacity=1)
        patch.set_stroke(width=0)
    coasts=VGroup()
    for ring in land_rings():
        if len(ring)<12: continue
        points=np.array([geo_point(*p,radius=radius+.025) for p in ring])
        curve=VMobject(stroke_color=COAST,stroke_width=1.0)
        def redraw(c,points=points):
            theta=camera.get_theta() if camera else -150*DEGREES
            phi=camera.get_phi() if camera else 82*DEGREES
            direction=np.array([np.sin(phi)*np.cos(theta),np.sin(phi)*np.sin(theta),np.cos(phi)])
            visible=points@direction>.035
            c.clear_points();previous=False
            for point,keep in zip(points,visible):
                if keep:
                    if previous: c.add_line_to(point)
                    else: c.start_new_path(point)
                previous=keep
            return c
        redraw(curve)
        if camera: curve.add_updater(redraw)
        coasts.add(curve)
    return VGroup(sphere,coasts)
