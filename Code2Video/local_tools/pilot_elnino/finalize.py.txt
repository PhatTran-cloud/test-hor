"""Export a continuous voice-only master from a completed local render snapshot."""
import argparse
import json
import subprocess
import sys
from pathlib import Path

import numpy as np
from PIL import Image, ImageDraw
from scipy.io import wavfile

sys.path.insert(0, str(Path(__file__).resolve().parents[2] / "src"))
from output_paths import get_output_root


def run(args):
    return subprocess.run(args, check=True, capture_output=True, encoding="utf-8", errors="replace")


def probe(path):
    return json.loads(run(["ffprobe", "-v", "error", "-show_streams", "-show_format",
                           "-of", "json", str(path)]).stdout)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("snapshot", type=Path)
    parser.add_argument("--output", type=Path, default=get_output_root() / "elnino")
    args = parser.parse_args()
    root, out = args.snapshot.resolve(), args.output.resolve()
    out.mkdir(parents=True, exist_ok=True)
    plan = json.loads((root / "plan.json").read_text(encoding="utf-8"))
    timing = json.loads((root / "speech_timing.json").read_text(encoding="utf-8"))
    parts, clips, timeline = [], [], []
    offset = 0
    for section in plan["sections"]:
        sid = section["id"]
        video = root / f"media_{sid}" / "videos" / sid / "1080p60" / f"Section{sid.split('_')[1]}Scene.mp4"
        meta = probe(video)
        stream = next(s for s in meta["streams"] if s["codec_type"] == "video")
        assert (stream["width"], stream["height"], stream["r_frame_rate"]) == (1920, 1080, "60/1")
        rate, samples = wavfile.read(root / "audio" / f"{sid}.wav")
        assert rate == 48000 and samples.ndim == 1
        frames = int(stream["nb_frames"])
        delta = frames * 800 - len(samples)
        if not 0 <= delta <= 1600:
            raise ValueError(f"Unexpected picture/speech difference in {sid}: {delta} samples")
        samples = np.pad(samples, (0, delta))
        parts.append(samples)
        clips.append("file '" + video.as_posix().replace("'", "'\\''") + "'")
        t = timing[sid]
        timeline.append({"id": sid, "title": section["title"], "start": offset / 48000,
                         "duration": len(samples) / 48000,
                         "speech_start": offset / 48000 + t["speech_start"],
                         "speech_end": offset / 48000 + t["speech_end"],
                         "voice": t["voice"], "speed": t["speed"], "time_stretch": False})
        offset += len(samples)
    soundtrack = out / "narration.wav"
    wavfile.write(soundtrack, 48000, np.concatenate(parts))
    concat = out / "picture_concat.txt"
    concat.write_text("\n".join(clips) + "\n", encoding="utf-8")
    picture = out / "picture_1080p60.mp4"
    run(["ffmpeg", "-v", "error", "-y", "-f", "concat", "-safe", "0", "-i", str(concat),
         "-c", "copy", str(picture)])
    final = out / "elnino_1080p60.mp4"
    run(["ffmpeg", "-v", "error", "-y", "-i", str(picture), "-i", str(soundtrack),
         "-map", "0:v:0", "-map", "1:a:0", "-c:v", "copy", "-c:a", "aac", "-b:a", "192k",
         "-af", "loudnorm=I=-16:TP=-1.5:LRA=11", "-ar", "48000", "-ac", "2",
         "-t", f"{offset / 48000:.9f}", "-movflags", "+faststart", str(final)])
    run(["ffmpeg", "-v", "error", "-i", str(final), "-f", "null", "-"])
    meta = probe(final)
    v = next(s for s in meta["streams"] if s["codec_type"] == "video")
    a = next(s for s in meta["streams"] if s["codec_type"] == "audio")
    joins = [timeline[i + 1]["speech_start"] - timeline[i]["speech_end"] for i in range(len(timeline) - 1)]
    assert max(joins) < 1
    assert (v["width"], v["height"], v["r_frame_rate"]) == (1920, 1080, "60/1")
    assert int(v["nb_frames"]) == offset // 800
    assert a["sample_rate"] == "48000" and a["channels"] == 2
    verification = {"snapshot": str(root), "video": str(final), "duration": offset / 48000,
                    "width": v["width"], "height": v["height"], "fps": v["r_frame_rate"],
                    "frames": int(v["nb_frames"]), "audio_rate": a["sample_rate"], "audio_channels": a["channels"],
                    "inter_scene_silence_seconds": joins, "full_decode": "passed", "subtitles": False,
                    "sfx": False, "music": False, "time_stretch": False}
    for name, content in [("verification.json", verification), ("audio_timeline.json", timeline),
                           ("sources.json", plan["sources"]),
                           ("sound_design.json", {"mode": "voice_only", "events": [], "music": None})]:
        (out / name).write_text(json.dumps(content, indent=2), encoding="utf-8")
    (out / "script.txt").write_text("\n\n".join(s["title"] + "\n" + s["narration"] for s in plan["sections"]), encoding="utf-8")
    # Review actual decoded frames at early/middle/late points, not source-only stills.
    sheet = Image.new("RGB", (1440, 6 * 290), "#081c25")
    draw = ImageDraw.Draw(sheet)
    for row, scene in enumerate(timeline):
        for col, fraction in enumerate((.15, .5, .85)):
            at = scene["start"] + scene["duration"] * fraction
            frame = out / f"qa_{row + 1}_{col + 1}.jpg"
            run(["ffmpeg", "-v", "error", "-y", "-ss", str(at), "-i", str(final),
                 "-frames:v", "1", "-vf", "scale=480:270", str(frame)])
            with Image.open(frame) as im:
                sheet.paste(im, (col * 480, row * 290))
            draw.text((col * 480 + 8, row * 290 + 273), f"{scene['id']} / {at:.2f}s", fill="white")
    sheet.save(out / "contact_sheet_final.jpg", quality=94)
    print(json.dumps(verification, indent=2))


if __name__ == "__main__":
    main()
