from elnino_scene_base import *

class Section1Scene(ClimateScene):
    def construct(self):
        self.header('WHEN THE OCEAN SHIFTS', 'EL NINO / FIRST, THE NORMAL PACIFIC')
        m=PacificMap(); m.land.set_z_index(2)
        regions=m.region_labels().set_z_index(5)
        self.play(FadeIn(m),FadeIn(regions),run_time=1.2)
        normal=label('NORMAL CONDITIONS',[0,2.3,0],MUTED,18).set_z_index(5)
        self.play(FadeIn(normal),run_time=.6)
        self.until(6)
        winds=wind_set(left=-1.0,right=3.2,y=1.35,direction=-1,count=6).set_z_index(4)
        wind_tag=label('TRADE WINDS: EAST TO WEST',[0,1.85,0],GOLD,19).set_z_index(5)
        self.play(LaggedStart(*[GrowArrow(a) for a in winds],lag_ratio=.08),FadeIn(wind_tag),run_time=1.4)
        wind_flow=flow_dots(Line([3.5,1.02,0],[-1.7,1.02,0]),GOLD,12,.09).set_z_index(4)
        self.add(wind_flow)
        self.until(11)
        warm=heat_ribbon(m,136,182,WARM).set_z_index(1)
        water=flow_dots(Line(m.xy(248,0),m.xy(136,0)),WARM,12,.055).set_z_index(3)
        warm_tag=label('WARM POOL',m.xy(163,-12),WARM,20).set_z_index(5)
        self.play(FadeIn(warm),FadeIn(water),FadeIn(warm_tag),run_time=2)
        self.until(16)
        cold=heat_ribbon(m,250,277,COLD,-3).set_z_index(1)
        cold_tag=label('COOLER WATER',m.xy(257,-14),COLD,19).set_z_index(5)
        self.play(FadeIn(cold),FadeIn(cold_tag),run_time=1.2)
        coast=Dot(m.xy(280,-5),color=COLD,radius=.045).set_z_index(5)
        self.play(FadeIn(coast),Flash(coast,color=COLD,flash_radius=.26),run_time=.8)
        self.until(21)
        c=cloud(m.xy(158,25),.86).set_z_index(6)
        rain=falling_rain(m.xy(158,21),.75,12).set_z_index(6)
        rising=flow_dots(Line(m.xy(158,2),m.xy(158,20)),RAIN,6,.12).set_z_index(6)
        self.play(FadeIn(c),FadeIn(rain),FadeIn(rising),run_time=1)
        self.until(26)
        self.footer('Wind. Water. Rain. One connected system.',ICE)
        self.finish_slot()
