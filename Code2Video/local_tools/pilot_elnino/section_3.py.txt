from elnino_scene_base import *

class Section3Scene(ClimateScene):
    def construct(self):
        self.header('WHAT HAPPENS WHEN THE WIND WEAKENS?', 'EL NINO / THE OCEAN AND ATMOSPHERE RESPOND TOGETHER')
        sea=OceanSlice()
        winds=wind_set(y=1.0,count=6)
        regions=VGroup(label('WEST',[-5.3,1.8,0],MUTED,19),label('EAST',[5.3,1.8,0],MUTED,19))
        self.add(sea,regions,winds)
        self.play(*[a.animate.scale(.36,about_point=a.get_start()).set_opacity(.28) for a in winds],run_time=2.3)
        tag=label('WEAKER TRADE WINDS',[0,1.7,0],GOLD,22)
        self.play(FadeIn(tag),run_time=.7)
        self.until(5)
        east=Arrow([-3.8,-.25,0],[4.8,-.25,0],color=ICE,stroke_width=3,buff=.06)
        dots=flow_dots(Line([-4.8,-.4,0],[4.8,-.4,0]),ICE,18,.07)
        self.play(GrowArrow(east),FadeIn(dots),run_time=1.3)
        self.until(9)
        self.play(sea.state.animate.set_value(1),run_time=7,rate_func=smooth)
        self.until(17)
        cold=Arrow([4.6,-2.8,0],[4.6,-1.7,0],color=COLD,stroke_width=2,buff=.03)
        barrier=DashedLine([3.6,-1.5,0],[5.5,-1.5,0],color=COLD,stroke_width=2)
        self.play(GrowArrow(cold),Create(barrier),run_time=1.1)
        label_cold=label('LESS EFFECTIVE COOLING',[2.4,-2.6,0],COLD,20)
        self.play(FadeIn(label_cold),run_time=.8)
        self.until(23)
        self.play(FadeOut(tag),run_time=.4)
        feedback=VGroup(label('WEAKER WINDS',[-2.35,1.95,0],GOLD,23),label('WARMER EAST',[2.35,1.95,0],HOT,23),DoubleArrow([-.8,1.95,0],[.85,1.95,0],color=ICE,buff=0,stroke_width=2))
        self.play(FadeIn(feedback),run_time=.8)
        self.until(27)
        self.footer('A reinforcing feedback, not a one-way switch.',MUTED)
        self.finish_slot()
