from elnino_scene_base import *

class Section2Scene(ClimateScene):
    def construct(self):
        self.header('THE OCEAN HAS A HIDDEN SLOPE', 'NORMAL PACIFIC / CROSS-SECTION ALONG THE EQUATOR')
        sea=OceanSlice()
        labels=VGroup(label('WEST: INDONESIA',[-4.5,1.8,0],MUTED,20),label('EAST: SOUTH AMERICA',[4.1,1.8,0],MUTED,20))
        self.play(FadeIn(sea),FadeIn(labels),run_time=1.3)
        scale_note=label('DEPTH EXAGGERATED',[-4.2,-2.96,0],MUTED,14)
        self.add(scale_note)
        self.until(5)
        warm_tag=label('WARM SURFACE WATER',[-1.4,-.3,0],INK,22)
        cold_tag=label('COLDER DEEP WATER',[-.6,-2.7,0],RAIN,20)
        self.play(FadeIn(warm_tag),FadeIn(cold_tag),run_time=1)
        self.until(10)
        thermo_tag=label('THERMOCLINE',[1.0,-2.23,0],COLD,22)
        leader=Line([2,-2.05,0],[3.0,sea.depth(3),0],color=COLD,stroke_width=1.2)
        self.play(FadeIn(thermo_tag),Create(leader),run_time=1)
        self.until(14)
        winds=wind_set(y=1.0,count=6)
        self.play(LaggedStart(*[GrowArrow(a) for a in winds],lag_ratio=.08),run_time=1.1)
        water=flow_dots(Line([4.8,-.05,0],[-4.8,-.05,0]),HOT,15,.065)
        self.add(water)
        depth=DoubleArrow([-5.25,.2,0],[-5.25,sea.depth(-5.25),0],buff=.05,color=INK,stroke_width=2)
        self.play(GrowArrow(depth),run_time=1)
        self.until(20)
        up=Arrow([4.6,-2.85,0],[4.6,.17,0],buff=.04,color=COLD,stroke_width=3)
        bubbles=flow_dots(Line([4.6,-2.8,0],[4.6,.2,0]),COLD,7,.17)
        self.play(GrowArrow(up),FadeIn(bubbles),run_time=1.3)
        self.until(25)
        self.footer('A shallow eastern thermocline helps cold water reach the surface.',MUTED)
        nutrients=VGroup(*[Dot([4.6+(i%3-1)*.2,-2.4-i*.07,0],radius=.035,color=GREEN) for i in range(9)])
        self.play(nutrients.animate.shift(UP*2.1),run_time=3,rate_func=linear)
        self.finish_slot()
