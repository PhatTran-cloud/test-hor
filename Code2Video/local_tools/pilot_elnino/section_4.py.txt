from elnino_scene_base import *

class Section4Scene(ClimateScene):
    def construct(self):
        self.header('THE RAIN BELT MOVES WITH THE HEAT', 'EL NINO / A SHIFT IN TROPICAL CONVECTION')
        m=PacificMap();m.land.set_z_index(2)
        regions=m.region_labels().set_z_index(5)
        west=heat_ribbon(m,136,180,WARM).set_z_index(1)
        self.add(m,regions,west)
        c=cloud(m.xy(158,25),1.0).set_z_index(6)
        rain=falling_rain(m.xy(158,20),1.0,14).set_z_index(6)
        rising=flow_dots(Line(m.xy(158,0),m.xy(158,19)),RAIN,6,.1).set_z_index(6)
        self.play(FadeIn(c),FadeIn(rain),FadeIn(rising),run_time=1.4)
        normal=label('NORMAL: RISING AIR IN THE WEST',[0,2.3,0],MUTED,18)
        self.add(normal)
        self.until(11)
        self.play(FadeOut(normal),FadeOut(rising),FadeOut(rain),run_time=.6)
        east=heat_ribbon(m,176,277,HOT).set_z_index(1)
        track=Line(m.xy(158,25),m.xy(235,25))
        self.play(ReplacementTransform(west,east),MoveAlongPath(c,track),run_time=6,rate_func=smooth)
        self.until(18)
        new_rain=falling_rain(m.xy(235,20),1.15,16).set_z_index(6)
        new_rising=flow_dots(Line(m.xy(235,0),m.xy(235,19)),RAIN,7,.11).set_z_index(6)
        self.play(FadeIn(new_rain),FadeIn(new_rising),run_time=.8)
        wetter=label('MORE RAIN',m.xy(235,-13),RAIN,20).set_z_index(5)
        drier=label('DRIER RISK',m.xy(140,-17),GOLD,20).set_z_index(5)
        self.play(FadeIn(wetter),FadeIn(drier),run_time=.9)
        self.until(24)
        signal=Line(m.xy(174,0),m.xy(270,0))
        self.add(flow_dots(signal,HOT,14,.05).set_z_index(4))
        self.footer('Shift the ocean heat, and the atmosphere shifts with it.',MUTED)
        self.finish_slot()
