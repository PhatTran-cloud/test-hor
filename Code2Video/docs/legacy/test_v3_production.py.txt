import copy
import json
import sys
import tempfile
import unittest
from pathlib import Path

import numpy as np
from scipy.io import wavfile

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / "src"))
from v3_contract import normalize_plan
from sfx import mix_narration_with_sfx, RATE, _read_stereo
from sfx_assets import commercial_license, prepare_cues


class ContractTests(unittest.TestCase):
    def setUp(self):
        self.plan = json.loads((ROOT / "local_tools/v3_smoke/plan.json").read_text())

    def test_legacy_preserves_route_and_voice_only(self):
        legacy = {"sections": [{"id": "section_1", "source": "scene.py"}]}
        result = normalize_plan(legacy)
        self.assertEqual(result["sections"][0]["renderer"], "manim")
        self.assertEqual(result["sound_design"], "voice_only")
        self.assertNotIn("renderer", legacy["sections"][0])

    def test_mixed_plan_and_adaptive_default(self):
        self.plan.pop("sound_design")
        result = normalize_plan(self.plan)
        self.assertEqual(result["sound_design"], "adaptive")
        self.assertEqual([s["renderer"] for s in result["sections"][0]["shots"]], ["manim", "remotion", "threejs"])

    def test_reject_wrong_engine_clock_and_incomplete_3d(self):
        for mutate in (
            lambda p: p.update(fps=30),
            lambda p: p["sections"][0]["shots"][0].update(renderer="unknown"),
            lambda p: p["sections"][0]["shots"][1]["render_spec"].update(time_model="wall_clock"),
            lambda p: p["sections"][0]["shots"][2]["three_contract"].pop("workflow"),
            lambda p: p["sections"][0]["shots"][1]["vfx_events"][0].update(end=4),
            lambda p: p["sections"][0].update(editorial={"evidence_type": "evidence", "sources": []}),
            lambda p: p["sections"][0].update(scientific_contract={"mode": "numerical_simulation"}),
        ):
            plan = copy.deepcopy(self.plan)
            mutate(plan)
            with self.assertRaises(ValueError):
                normalize_plan(plan)
        bad_gain = copy.deepcopy(self.plan)
        bad_gain["sections"][0]["sfx_cues"] = [{"trigger": "x", "purpose": "y", "gain_db": 2}]
        with self.assertRaises(ValueError):
            normalize_plan(bad_gain)


class SoundTests(unittest.TestCase):
    def test_asset_mix_is_stereo_ducked_deterministic_and_does_not_clip(self):
        with tempfile.TemporaryDirectory() as directory:
            root = Path(directory)
            t = np.arange(RATE * 3) / RATE
            speech = .15 * np.sin(2 * np.pi * 180 * t)
            speech[:RATE] = 0
            speech[2 * RATE:] = 0
            asset = np.column_stack([.2 * np.sin(2 * np.pi * 600 * t), .1 * np.sin(2 * np.pi * 900 * t)])
            wavfile.write(root / "voice.wav", RATE, speech.astype(np.float32))
            wavfile.write(root / "effect.wav", RATE, asset.astype(np.float32))
            cues = [{"file": str(root / "effect.wav"), "trigger": "Persistent machinery",
                "purpose": "Spatial context", "role": "ambience", "relative_time": 0,
                "duration": 3, "gain_db": -18, "fade_in": .01, "fade_out": .01}]
            prepared = prepare_cues(cues, root)
            for name in ("a", "b"):
                mix_narration_with_sfx(root / "voice.wav", root / f"{name}.wav", prepared)
            self.assertEqual((root / "a.wav").read_bytes(), (root / "b.wav").read_bytes())
            _, mix = wavfile.read(root / "a.wav")
            self.assertEqual(mix.shape, (RATE * 3, 2))
            self.assertLess(float(np.max(np.abs(mix))), .96)
            effects = mix - _read_stereo(root / "voice.wav")
            quiet = np.sqrt(np.mean(effects[RATE // 4:RATE // 2] ** 2))
            spoken = np.sqrt(np.mean(effects[RATE + RATE // 4:RATE + RATE // 2] ** 2))
            self.assertLess(spoken, quiet * .5)
            self.assertFalse(np.array_equal(mix[:, 0], mix[:, 1]))
            ledger = json.loads((root / "a.sfx.json").read_text())
            self.assertEqual(ledger["listening_status"], "UNVERIFIED")
            self.assertTrue(ledger["cues"][0]["provenance"]["sha256"])
            bad = [{**prepared[0], "relative_time": 2}]
            with self.assertRaises(ValueError):
                mix_narration_with_sfx(root / "voice.wav", root / "bad.wav", bad)
            with self.assertRaises(ValueError):
                mix_narration_with_sfx(root / "voice.wav", root / "missing.wav", [{k: v for k, v in cues[0].items() if k != "file"}])

    def test_commercial_asset_policy(self):
        self.assertTrue(commercial_license("https://creativecommons.org/publicdomain/zero/1.0/"))
        self.assertTrue(commercial_license("https://creativecommons.org/licenses/by/4.0/"))
        self.assertFalse(commercial_license("https://creativecommons.org/licenses/by-nc/4.0/"))

    def test_v2_procedural_cue_migrates_explicitly(self):
        with tempfile.TemporaryDirectory() as directory:
            cue = {"type": "reveal", "trigger": "Legacy reveal", "purpose": "Compatibility", "relative_time": 0, "duration": .2}
            self.assertEqual(prepare_cues([cue], Path(directory), allow_legacy=True)[0]["source"], "procedural")
            with self.assertRaises(ValueError):
                prepare_cues([cue], Path(directory))


if __name__ == "__main__":
    unittest.main()
